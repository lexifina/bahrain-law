وزارة التجارة والزراعة

قرار رقم (3) لسنة 1993

بشأن الرسوم التي تُحَصَّـل تنفيذاً

للمرسوم بقانون رقم (10) لسنة 1992

بشأن الوكالة التجارية

وزير التجارة والزراعة:

بعد الاطلاع على المادة (31) من المرسوم بقانون رقم (10) لسنة 1992 بشأن الوكالة التجارية،

وبعد موافقة مجلس الوزراء،

قرر الآتي:

مادة - 1 -

تـُحَصَّـل الرسوم المبينة فيما يلي عن الطلبات الآتية:

الطلب الرسم

1- طلب القيــــــــــد عـــــــــــن كــــــــــــل وكالــــــــــــــة تجاريـــــــــــــــــــة 20 ديناراً

2- طلب تجديد قيــــــد وكالــــــــــة تجاريـــــة عـــــــن كـــــل سنتيــــــــــــــــن 20 ديناراً

3- طلب إجراء تأشير بإضافة أو تعديل أو شطب في البـيانات المدونة في السجل 5 دنانيـــر

4- طلب شهادة رسمية عن بيان مدون بالسجل أو طلب صورة طبق الأصــــــل 2 ديناران

مـــــــــــــــــن عقـــــــــــــــــــــد الوكالــــــــــــــــــــة التجاريـــــــــــــــــــة.

5- طلـــب استـفســـــــار عن أي بيــان يتعلق بوكالـــــــــــة تجاريــــــــــــــــــة 2 ديناران

6- طلــــب الحصـــــول علـــــى نسخـــــة مــــــن المرســوم بقانــون بشــــــأن 2 ديناران

الوكالــــــــة التجاريــــــــــة ولائحتــــــــــه التـنفيذيـــــــــــــــــــــــــــــــــــة

7- طلـــــب إصــــــــــدار شهــــــــادة للجمارك بالإفســـاح عــــن بضاعــــــــة 2 ديناران

مادة - 2 -

على وكيل وزارة التجارة والزراعة تـنفيذ هذا القرار، ويعمل به من تاريخ نشره في الجريدة الرسمية.
